CREATE TABLE NguoiDung (
    maNguoiDung VARCHAR(50) PRIMARY KEY,
    tenDangNhap VARCHAR(50),
    matKhauMaHoa VARCHAR(255),
    email VARCHAR(100),
    soDienThoai VARCHAR(20),
    hoTen VARCHAR(100),
    vaiTro VARCHAR(20),
    daXacThuc BOOLEAN,
    trangThai VARCHAR(20),
    ngayTao TIMESTAMP
);

CREATE TABLE NhaCungCap (
    maNhaCungCap VARCHAR(50) PRIMARY KEY,
    tenCongTy VARCHAR(200),
    maSoThue VARCHAR(50),
    soDienThoaiLienHe VARCHAR(20),
    emailLienHe VARCHAR(100),
    diemUyTin DOUBLE PRECISION
);

CREATE TABLE DanhMuc (
    maDanhMuc VARCHAR(50) PRIMARY KEY,
    tenDanhMuc VARCHAR(200),
    moTa TEXT
);

CREATE TABLE DiaChiGiaoHang (
    maDiaChi VARCHAR(50) PRIMARY KEY,
    tenNguoiNhan VARCHAR(100),
    soDienThoaiNguoiNhan VARCHAR(20),
    diaChiChiTiet VARCHAR(255),
    viDo DOUBLE PRECISION,
    kinhDo DOUBLE PRECISION,
    macDinh BOOLEAN,
    maNguoiDung VARCHAR(50),
    CONSTRAINT fk_diachi_nguoidung FOREIGN KEY (maNguoiDung) REFERENCES NguoiDung(maNguoiDung)
);

CREATE TABLE VuonUom (
    maVuonUom VARCHAR(50) PRIMARY KEY,
    tenVuonUom VARCHAR(200),
    diaChi VARCHAR(255),
    viDo DOUBLE PRECISION,
    kinhDo DOUBLE PRECISION,
    banKinhGiaoToiDaKm DOUBLE PRECISION,
    nguongTonKhoToiThieu INT,
    danhGiaNhaCungCap DOUBLE PRECISION,
    trangThai VARCHAR(20),
    maNhaCungCap VARCHAR(50),
    CONSTRAINT fk_vuonuom_nhacungcap FOREIGN KEY (maNhaCungCap) REFERENCES NhaCungCap(maNhaCungCap)
);

CREATE TABLE CayCanh (
    maCay VARCHAR(50) PRIMARY KEY,
    tenCay VARCHAR(200),
    tenKhoaHoc VARCHAR(200),
    moTaKichThuoc VARCHAR(100),
    giaBan NUMERIC(15, 2),
    moTa TEXT,
    huongDanChamSoc TEXT,
    khoiLuongKg DOUBLE PRECISION,
    trangThai VARCHAR(20),
    daXuatBan BOOLEAN,
    maDanhMuc VARCHAR(50),
    CONSTRAINT fk_caycanh_danhmuc FOREIGN KEY (maDanhMuc) REFERENCES DanhMuc(maDanhMuc)
);

CREATE TABLE YeuCauSinhTruong (
    maYeuCau VARCHAR(50) PRIMARY KEY,
    nhietDoToiThieu DOUBLE PRECISION,
    nhietDoToiDa DOUBLE PRECISION,
    doAmToiUu DOUBLE PRECISION,
    yeuCauAnhSang VARCHAR(100),
    loaiDat VARCHAR(100),
    maCay VARCHAR(50) NOT NULL UNIQUE,
    CONSTRAINT fk_yeucau_caycanh FOREIGN KEY (maCay) REFERENCES CayCanh(maCay)
);

CREATE TABLE TonKho (
    maTonKho VARCHAR(50) PRIMARY KEY,
    soLuongTon INT,
    soLuongTamGiu INT,
    lanCapNhatCuoi TIMESTAMP,
    maVuonUom VARCHAR(50),
    maCay VARCHAR(50),
    CONSTRAINT fk_tonkho_vuonuom FOREIGN KEY (maVuonUom) REFERENCES VuonUom(maVuonUom),
    CONSTRAINT fk_tonkho_caycanh FOREIGN KEY (maCay) REFERENCES CayCanh(maCay)
);

CREATE TABLE TaptinTruyenThong (
    maTruyenThong VARCHAR(50) PRIMARY KEY,
    loaiTruyenThong VARCHAR(30),
    duongDanTaptin VARCHAR(500),
    dungLuongByte BIGINT,
    laAnhChinh BOOLEAN,
    maCay VARCHAR(50),
    CONSTRAINT fk_taptin_caycanh FOREIGN KEY (maCay) REFERENCES CayCanh(maCay)
);

CREATE TABLE DonHang (
    maDonHang VARCHAR(50) PRIMARY KEY,
    ngayDatHang TIMESTAMP,
    tongTien NUMERIC(15, 2),
    phiVanChuyen NUMERIC(15, 2),
    khoangCachKm DOUBLE PRECISION,
    phuongThucThanhToan VARCHAR(50),
    trangThaiThanhToan VARCHAR(20),
    trangThaiDonHang VARCHAR(20),
    tamGiuDen TIMESTAMP,
    daXacNhanSucKhoe BOOLEAN,
    anhSucKhoeUrl VARCHAR(500),
    maNguoiDung VARCHAR(50),
    maDiaChi VARCHAR(50),
    maVuonUom VARCHAR(50),
    CONSTRAINT fk_donhang_nguoidung FOREIGN KEY (maNguoiDung) REFERENCES NguoiDung(maNguoiDung),
    CONSTRAINT fk_donhang_diachi FOREIGN KEY (maDiaChi) REFERENCES DiaChiGiaoHang(maDiaChi),
    CONSTRAINT fk_donhang_vuonuom FOREIGN KEY (maVuonUom) REFERENCES VuonUom(maVuonUom)
);

CREATE TABLE ChiTietDonHang (
    maChiTietDonHang VARCHAR(50) PRIMARY KEY,
    soLuong INT,
    donGia NUMERIC(15, 2),
    thanhTien NUMERIC(15, 2),
    maDonHang VARCHAR(50),
    maCay VARCHAR(50),
    CONSTRAINT fk_chitiet_donhang FOREIGN KEY (maDonHang) REFERENCES DonHang(maDonHang),
    CONSTRAINT fk_chitiet_caycanh FOREIGN KEY (maCay) REFERENCES CayCanh(maCay)
);

CREATE TABLE DanhGia (
    maDanhGia VARCHAR(50) PRIMARY KEY,
    soSaoDanhGia INT,
    binhLuan TEXT,
    duongDanTruyenThong VARCHAR(500),
    ngayTao TIMESTAMP,
    maDonHang VARCHAR(50) NOT NULL UNIQUE,
    maChiTietDonHang VARCHAR(50),
    CONSTRAINT fk_danhgia_donhang FOREIGN KEY (maDonHang) REFERENCES DonHang(maDonHang),
    CONSTRAINT fk_danhgia_chitiet FOREIGN KEY (maChiTietDonHang) REFERENCES ChiTietDonHang(maChiTietDonHang)
);
